\section*{Gu\'ia Pr\'actica 4 -- Complejidad}

\begin{enumerate}
	\item Mostrar usando la definición de $\mathcal{O}()$ tomada de \footnote{Marzal y Gracia. Introducción al análisis de algoritmos.} \footnote{\href{https://arco.esi.uclm.es/public/mirror/Introducci\%C3\%B3n\%20al\%20an\%C3\%A1lisis\%20de\%20algoritmos.pdf}{https://arco.esi.uclm.es/public/mirror/Introducci\%C3\%B3n\%20al\%20an\%C3\%A1lisis\%20de\%20algoritmos.pdf}} : \\
	\vspace*{-7mm}
	\begin{center}
		$\mathcal{O}(f(n)) = $ \{ $t: \mathbb{N} \rightarrow \mathbb{R}^+$ $|$  $\exists c \in \mathbb{R}^+, 
		\exists n_0 \in \mathbb{N} : \forall n \geq n_0$  $\Rightarrow$   $t(n) \leq c f(n)$ \}.
	\end{center}
	
		\begin{enumerate}[a)]
			\item $50n-8 \in \mathcal{O}(n)$ 
			\item $5n^2-3n + 6 \in \mathcal{O}(n^2)$
			\item $2^n + 7n + 3 \in \mathcal{O}(2^n)$ 			
		\end{enumerate}
		
	\item Determinar $\mathcal{O}()$ y $\Omega()$ de las siguientes funciones:
		\begin{enumerate}[a)]
			\item $5n^2$ 
			\item $3n^2 + 2n$ 			
			\item $3^n + 2^n$ 
			\item $n + \log_2(n)$ 			
			\item $n^2 + \log_2(n)$ 			
		\end{enumerate}	
		
	\item Determinar $\Theta()$ de las siguientes funciones:
		\begin{enumerate}[a)]
			\item $n^2 + 2n$ 
			\item $t(n) = \twopartdef { 2n + 4 } {$ n  par $} {7n^2 + n} {$n impar$}$
			\item $t(n) = \twopartdef { 4n + 5 } {$ n  par $} {5n + n} {$n impar$}$
		\end{enumerate}	
		
	\item Reescribir los siguientes órdenes de complejidad de menor a mayor ($<$):\\
	$\mathcal{O}(n), \mathcal{O}(n\log n), \mathcal{O}(n!), \mathcal{O}(n^2), \mathcal{O}(5),
	\mathcal{O}(\log\log n), \mathcal{O}(3^n), \mathcal{O}(\sqrt{n}), \mathcal{O}(n^n),$ \\
	 $\mathcal{O}(n^3), \mathcal{O}(2^n), \mathcal{O}(\log n)$. 
	 
	 Etiquete cada $\mathcal{O}$ como
	 \textbf{sublineal}, \textbf{lineal} o \textbf{superlineal}. Dentro de éstas últimas puede
	 decir cuáles son \textbf{polinómicas} (cuadráticas, cúbicas, etc.) o \textbf{exponenciales}.
		
		
	\item Para cada uno de los fragmentos de código siguientes:
	\vspace{-3mm}
	\begin{itemize}
		\item Diga qué $ \mathcal{O}()$ poseen.
		\item Impleméntelos y dé el tiempo de ejecución para $\neq$ valores de $n$.
		\item Compare estimación teórica (primer ítem) con tiempos reales (segundo ítem).
	\end{itemize}

		ese número faltante en $\mathcal{O}(n)$. Sólo es posible utilizar $\mathcal{O}(1)$ espacio
		adicional además del arreglo $A$ mismo.\\
		Ejemplo:\\
		$N = 5$. El arreglo puede contener elementos (no repetidos) $\in [0, 4]$.\\
		Si el vector $=[1, 4, 2, 0]$, entonces el elemento que falta es \textbf{3}.
