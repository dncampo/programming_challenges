	\item Mostrar usando la definición de $\mathcal{O}()$ tomada de \footnote{Marzal y Gracia. Introducción al análisis de algoritmos.} \footnote{\href{https://arco.esi.uclm.es/public/mirror/Introducci\%C3\%B3n\%20al\%20an\%C3\%A1lisis\%20de\%20algoritmos.pdf}{https://arco.esi.uclm.es/public/mirror/Introducci\%C3\%B3n\%20al\%20an\%C3\%A1lisis\%20de\%20algoritmos.pdf}} : \\
	\vspace*{-7mm}
	\begin{center}
		$\mathcal{O}(f(n)) = $ \{ $t: \mathbb{N} \rightarrow \mathbb{R}^+$ $|$  $\exists c \in \mathbb{R}^+, 
		\exists n_0 \in \mathbb{N} : \forall n \geq n_0$  $\Rightarrow$   $t(n) \leq c f(n)$ \}.
	\end{center}
	
		\begin{enumerate}[a)]
			\item $t(n) = 50n-8 \in \mathcal{O}(n)$ 
			\item $t(n) = 5n^2-3n + 6 \in \mathcal{O}(n^2)$
			\item $t(n) = 2^n + 7n + 3 \in \mathcal{O}(2^n)$ 			
		\end{enumerate}
		
	\item Determinar $\mathcal{O}()$ y $\Omega()$ de las siguientes funciones:
		\begin{enumerate}[a)]
			\item $t(n) = 5n^2$ 
			\item $t(n) = 3n^2 + 2n$ 			
			\item $t(n) = 3^n + 2^n$ 
			\item $t(n) = n + \log_2(n)$ 			
			\item $t(n) = n^2 + \log_2(n)$ 			
		\end{enumerate}	
		
	\item Determinar $\Theta()$ de las siguientes funciones:
		\begin{enumerate}[a)]
			\item $t(n)=n^2 + 2n$ 
			\item \begin{equation*}
    t(n) = \begin{cases}
               2n + 4               & $n $ par\\
               7n^2 + n               & $n $ impar\\
           \end{cases}
		\end{equation*}
			\item \begin{equation*}
    t(n) = \begin{cases}
               4n + 5               & $n $ par\\
               5n + 0.5n               & $n $ impar\\
           \end{cases}
		\end{equation*}

%			\item $t(n) = \twopartdef { 4n + 5 } {$ n  par $} {5n + n} {$n impar$}$
		\end{enumerate}	
		
	\item Reescribir los siguientes órdenes de complejidad de menor a mayor ($<$):\\
	$\mathcal{O}(n), \mathcal{O}(n\log n), \mathcal{O}(n!), \mathcal{O}(n^2), \mathcal{O}(5),
	\mathcal{O}(\log\log n), \mathcal{O}(3^n), \mathcal{O}(\sqrt{n}), \mathcal{O}(n^n),$ \\

	\item Given array of integers, find the number of sorted pairs formed by its elements such that one element in the pair is divisible by the other one.

\textbf{Example}

For \textit{sequence = [1, 3, 2]}, the output should be
\textit{divisorsPairs(sequence) = 2}.

These pairs are: \textit{(1, 3), (1, 2)}.
	
	\begin{lstlisting}[language=C++, frame=single]
int divisorsPairs(std::vector<int> sequence) {

  int result = 0;

  for (int i = 0; i < sequence.size(); i++) {
    for (int j = i + 1; j<  sequence.size(); j++) {
      if (sequence[i] % sequence[j] == 0 && sequence[j] % sequence[i] == 0) {
        result++;
      }
    }
  }

  return result;
}

	\end{lstlisting}

	\item Given an array of integers, find the rightmost round number (it ends with one or more zeroes) in it and output its position in the array (0-based). If there are \textit{no round numbers} in the given array, output \textit{-1}.

int rightmostRoundNumber(std::vector<int> inputArray) {

  }

	\item Given an array of integers \textit{inputArray} and an integer \textit{maxSum}, for each \textit{i} (\textit{0} $\leq$ \textit{i $<$ length of inputArray}) find the rightmost \textit{j (j < length of inputArray)} such that the sum of all elements between the $i^{th}$ and $j^{th}$ \emph{elements} (both inclusive) is not greater than \textit{maxSum}.

\textbf{Example}

For \textit{inputArray} $= [2, 3, 0, 1, 2]$ and \textit{maxSum} $= 4$, the output should be
maximalAllowableSubarrays(inputArray, maxSum) $= [0, 3, 4, 4, 4]$.


\begin{lstlisting}[language=C++, frame=single]
std::vector<int> maximalAllowableSubarrays(std::vector<int> inputArray, int maxSum) {
  std::vector<int> right(inputArray.size());
  int j = -1;
  int curSum = 0;
  for (int i = 0; i < inputArray.size(); i++) {
    if (i > 0) {
      curSum -= inputArray[i - 1];
    }
    while (j + 1 < inputArray.size() && curSum + inputArray[j + 1] < maxSum) {
      curSum += inputArray[++j];
    }
    right[i] = j;
  }

  return right;
}
\end{lstlisting}
